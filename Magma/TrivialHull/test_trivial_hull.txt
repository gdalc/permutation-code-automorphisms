function sample_perm_order(k)
	//Samples permutation if length n = 2*k and with order k. This is done by sampling
    	//two disjoint cycles of length k. The cycles are joined to create a string, then the string is used to create a permutation
    	//of length n = 2*k
	S2k := Sym(2*k);
	//generate a permutation of S_{2*k} and write it in sequence form
	P := Eltseq(Random(S2k)); 
	cycles:=[];
	for i in [1..k-1] do
		cycles[P[i]] := P[i+1];
	end for;
	cycles[P[k]]:=P[1];

	for i in [1..k-1] do
		cycles[P[k+i]]:=P[k+i+1];
	end for;
	cycles[P[2*k]]:=P[k+1];


	perm:=S2k!cycles;
return perm, PermutationMatrix(Integers(),perm);
end function;


function gen_code(Fq, n, k)
	//Generate random code whose automorphism group contains P (and its powers).
	//P is sampled as a permutation of length n and order k
	ok := 0;
	while (ok eq 0) do
		cycles, P := sample_perm_order(k);
		P:=ChangeRing(P, Fq);
		G := ZeroMatrix(Fq, k, n);
		a := Vector(Fq, [Random(Fq) : i in [1..n]]);
		G[1]:=a;
		for i in [2..k] do
            		a := a*P;
			G[i]:=a;
		end for;
		if (Rank(G) eq k) then
              		ok := 1;
		end if;
	end while;

	S := RandomMatrix(Fq,k,k);
	while (Rank(S) lt k) do
		S := RandomMatrix(Fq,k,k);
	end while;
	return cycles, P, S*G;
end function;

   	while (num lt num_test) do

        	if (h eq 0) then
			num:=num+1;
		
			//our reduction
			t1:= Cputime();
			A :=Transpose(G)*(G*Transpose(G))^-1*G; //adjacency matrix
			//directed graph geneation
			edges:=[];
			weights:=[];
			for i in [1..n] do
				for j in [1..n] do
					a:= Integers()!A[i][j];
					if a ne 0 then
						Append(~edges,[i,j]);
						Append(~weights,a);
					end if;
				end for;
			end for;
			graph := Digraph< n | edges >;
			auto_group :=  AutomorphismGroup(graph);

			//validity check

				print "PROBLEM";
			end if;
		end if;
	end while;
